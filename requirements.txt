# Core runtime — everything the app needs to run, fully offline.
# These are pure-Python or ship prebuilt wheels, so the install is small and
# reliable on a fresh machine.
#
# Optional heavy features (on-device voice transcription, Google Workspace
# sync) are NOT here — they live in requirements-optional.txt so the base
# install stays fast and can't stall on a large PyTorch download. The app is
# fully functional without them.

# NOTE: `cryptography` was pinned here but never imported anywhere in the app,
# which implied an at-rest encryption feature that does not exist. The database
# and uploads are plaintext on disk; full-disk encryption (BitLocker/FileVault)
# is the control that protects them, and the installers and README now say so
# outright. Removed rather than left as a false signal. Google Workspace sync
# pulls it in transitively when you install requirements-optional.txt.
python-dateutil==2.9.0
openpyxl==3.1.5

# what reads it. Pure Python, no build step, keeps the offline install simple.
xlrd==2.0.2
pytz==2024.1
requests==2.32.3
PyPDF2==3.0.1
